# Decision system for inviting volunteers from different countries to microstates, allowing them to build some army.

# The total number of volunteers from a specific continent is calculated using the following formula:

# foreign_volunteers = SUM(THIS.manpower_k * 10 * THIS.party_popularity@[ideology])

# ROOT is the country that takes a decision
# THIS is a country from which manpower is taken

# THIS.manpower_k is manpower in thousands
# THIS.party_popularity@[ideology] is popularity of party, is the popularity of the political party, where ideology = ROOT.ruling_party
# (so that countries that share the same ideology as ROOT will contribute more volunteers relative to their population
# (e.g. a communist country will receive more volunteers from other communist countries))

# The resulting value is subtracted from THIS country's manpower and added to foreign_volunteers.
# This calculation is performed for each country on the continent that exists and is not at war (excluding ROOT).
# In the end, foreign_volunteers is added to ROOT's manpower.

invite_foreign_volunteers_cat = {
	invite_volunteers_from_europe = {
		icon = eng_propaganda_campaigns

		cost = 25

		available = {
			is_microstate = yes
			has_manpower < 10000
			any_other_country = {
				has_war = no
				capital_scope = { is_on_continent = europe }
			}
		}

		visible = { }

		fire_only_once = no
		days_remove = 30
		days_re_enable = 90

		remove_effect = {
			hidden_effect = {
				set_temp_variable = { foreign_volunteers = 0 }
				for_each_scope_loop = {
					array = global.countries
					if = {
						limit = {
							exists = yes
							capital_scope = { is_on_continent = europe }
							has_war = no
							NOT = { tag = ROOT }
						}
						set_temp_variable = { volunteers_from_country = THIS.manpower_k }
						multiply_temp_variable = { volunteers_from_country = 10 }
						if = {
							limit = {
								ROOT = { has_government = democratic }
							}
							multiply_temp_variable = { volunteers_from_country = THIS.party_popularity@democratic }
						}
						else_if = {
							limit = {
								ROOT = { has_government = communism }
							}
							multiply_temp_variable = { volunteers_from_country = THIS.party_popularity@communism }
						}
						else_if = {
							limit = {
								ROOT = { has_government = fascism }
							}
							multiply_temp_variable = { volunteers_from_country = THIS.party_popularity@fascism }
						}
						else_if = {
							limit = {
								ROOT = { has_government = neutrality }
							}
							multiply_temp_variable = { volunteers_from_country = THIS.party_popularity@neutrality }
						}
						multiply_temp_variable = { volunteers_from_country = -1 }
						add_manpower = var:volunteers_from_country
						multiply_temp_variable = { volunteers_from_country = -1 }
						add_to_temp_variable = { foreign_volunteers = var:volunteers_from_country }
					}
				}
				add_manpower = var:foreign_volunteers
			}
			custom_effect_tooltip = ems_volunteers_from_europe
		}

		ai_will_do = {
			base = 0
			modifier = {
				add = 10
				has_manpower < 10000
				OR = {
					has_government = communism
					has_government = fascism
				}
			}
		}
	}
